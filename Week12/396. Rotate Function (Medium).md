```c++
class Solution {
public:
    int maxRotateFunction(vector<int>& nums) {
        int n = nums.size();
        int f = 0;
        
        int numSum = 0;
        for (int num: nums) {
            numSum += num; 
        }
        
        // calculate the initial F(0)
        for (int i = 0; i < n; i++) {
            f += i * nums[i];
        }
        int res = f;
        // each rotation adds the sum of all elements, and subtracts n * the element that rotates to the front
        // F(i) = F(i-1) + sum - n * nums[n - i]
        for (int i = n - 1; i > 0; i--) {
            f += numSum - n * nums[i];
            res = max(res, f);
        }
        return res;
    }
};
```
