class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        // If total gas is less than total cost, we can never complete the circuit.
        // If we start at station A and run out of gas at station B, we can't start at any station between A and B either,
        // because starting from any station in between will only give us less gas.
        // So we just reset our starting station to B + 1 and reset our gas tank to 0.

        int n = gas.size();
        int totalTank = 0;
        int tempTank = 0;
        int startFrom = 0;
        
        for(int i = 0; i < n; i++){
            int diff = gas[i] - cost[i];
            totalTank += diff;
            tempTank += diff;
            if(tempTank < 0){
                tempTank = 0;
                startFrom = i + 1;
                // printf("start again from %d", startFrom);
            }
        }
        // we still have to check if total is positive
        return (totalTank < 0) ? -1 : startFrom;
    }
};
